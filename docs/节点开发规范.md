## 概述

### 什么是节点？

节点是 NodeFlow 框架中的基本计算单元，代表一个独立的进程。每个节点通过 Unix Domain Socket 与其他节点进行数据交互。

### 节点特性

- **独立进程**：每个节点作为单独的 Python 子进程运行
- **声明式配置**：通过 YAML 文件声明端口和参数
- **进程间通信**：基于 Unix Domain Socket 的 JSON 消息传输
- **非阻塞接收**：使用 `recv_latest()` 获取最新值（不排队）
- **参数化**：通过环境变量传递配置给节点进程

### 节点的三个角色

1. **源节点** (Source) - 仅有输出，无输入
2. **处理节点** (Processing) - 既有输入又有输出
3. **汇节点** (Sink) - 仅有输入，无输出

## node.yaml 规范

### 完整模板

```yaml
# 基本信息
name: node_name                    # 节点包名（必需，英文，唯一）
version: "1.0.0"                   # 版本号（推荐语义化版本）
description: "节点描述"             # 简短描述（必需）

# 启动入口
entrypoints:                       # 各平台的启动方式（必需）
  linux:
    kind: python                   # 运行环境（必需：python）
    cmd: ["python3", "run.py"]     # 启动命令（必需）
  darwin:
    kind: python
    cmd: ["python3", "run.py"]
  windows:
    kind: python
    cmd: ["python", "run.py"]

# 端口定义
ports:                             # 端口集合（必需）
  inputs:                          # 输入端口列表（可为空）
    - name: input_name
      type: json                   # 数据类型（json, any）
      description: "输入说明"

  outputs:                         # 输出端口列表（可为空）
    - name: output_name
      type: json
      description: "输出说明"

# 参数定义
params:                            # 参数字典（可为空）
  param_name:                      # 参数名
    type: string                   # 参数类型（string, number, bool）
    default: "default_value"       # 默认值（可选）
    required: false                # 是否必需（可选）
    description: "参数说明"
```

### 重要规则

#### ❌ 常见错误 - YAML 格式错误

```yaml
# ❌ 错误：params 不能是列表！
params:
  - name: param1
    type: string
  - name: param2
    type: number

# ✅ 正确：params 是字典
params:
  param1:
    type: string
  param2:
    type: number
```

#### ❌ 常见错误 - 端口格式错误

```yaml
# ❌ 错误：inputs/outputs 不能在顶级
inputs:
  - name: input1
outputs:
  - name: output1

# ✅ 正确：必须在 ports 内
ports:
  inputs:
    - name: input1
  outputs:
    - name: output1
```

#### ❌ 常见错误 - 缺少 entrypoints 的 kind 字段

```yaml
# ❌ 错误：缺少 kind
entrypoints:
  linux:
    cmd: ["python3", "run.py"]

# ✅ 正确：必须指定 kind
entrypoints:
  linux:
    kind: python
    cmd: ["python3", "run.py"]
```

## run.py 实现规范

### 标准框架

```python
#!/usr/bin/env python3
"""
节点名称：<节点名>

功能描述：
<详细的中文说明>

输入端口：
- <端口名>: <数据格式说明>

输出端口：
- <端口名>: <数据格式说明>

可配置参数：
- <参数名> (类型): <说明> [default: 默认值]
"""

import time
import json
from pathlib import Path
import sys

# 添加 SDK 路径（相对于 node-hub/node_name/）
sys.path.insert(0, str(Path(__file__).parent.parent.parent))

from sdk.nodeflow_sdk import NodeFlowSDK


def main():
    """节点主函数"""
    # 初始化 SDK (context manager)
    with NodeFlowSDK(log_level="INFO") as sdk:
        sdk.logger.info(f"节点启动: {sdk.node_id}")

        # 1. 读取参数
        param1 = sdk.get_param('param1', 'default_value')
        param2 = sdk.get_param('param2', 100)

        sdk.logger.info(f"配置: param1={param1}, param2={param2}")

        # 2. 创建端口
        # 输入端口（可选）
        if True:  # 检查是否需要输入
            input_port = sdk.create_input_port('input_name')

        # 输出端口（可选）
        if True:  # 检查是否需要输出
            output_port = sdk.create_output_port('output_name')

        sdk.logger.info("端口创建完成")

        # 3. 初始化状态变量
        seq = 0
        start_time = time.time()

        # 4. 主循环
        try:
            while True:
                current_time = time.time()

                # 4a. 接收输入（非阻塞）
                if 'input_port' in locals():
                    input_data = input_port.recv_latest()
                    if input_data is not None:
                        sdk.logger.debug(f"接收数据: {input_data}")
                else:
                    input_data = None

                # 4b. 处理数据
                output_data = {
                    'timestamp': current_time,
                    'seq': seq,
                    'data': 'processed',
                    # ... 其他字段
                }

                # 4c. 发送输出
                if 'output_port' in locals():
                    output_port.send(output_data)

                # 4d. 定期日志
                if seq % 10 == 0:
                    elapsed = time.time() - start_time
                    sdk.logger.info(f"运行 {seq} 个周期，耗时 {elapsed:.1f}s")

                seq += 1

                # 休眠（根据频率调整）
                time.sleep(0.1)

        except KeyboardInterrupt:
            sdk.logger.info("收到关闭信号")
        except Exception as e:
            sdk.logger.error(f"错误: {e}", exc_info=True)
        finally:
            sdk.logger.info(f"节点停止，共处理 {seq} 个周期")


if __name__ == '__main__':
    main()
```

### 关键点

#### 1. SDK 初始化

```python
# ✅ 正确：使用 context manager
with NodeFlowSDK(log_level="INFO") as sdk:
    sdk.logger.info("节点启动")
    # ... 节点代码

# ❌ 错误：直接初始化
sdk = NodeFlowSDK()  # 资源可能未正确清理
```

#### 2. 参数读取

```python
# ✅ 正确：使用 get_param 获取参数
param = sdk.get_param('param_name', default_value)

# ❌ 错误：直接访问环境变量
param = os.environ.get('NODE_param_name')  # 不规范
```

#### 3. 端口创建

```python
# ✅ 正确：先检查端口是否存在（可选）
input_port = sdk.create_input_port('input_name')
output_port = sdk.create_output_port('output_name')

# ❌ 错误：端口名与 node.yaml 不匹配
output_port = sdk.create_output_port('wrong_name')  # 运行时错误
```

#### 4. 数据接收（非阻塞）

```python
# ✅ 正确：使用 recv_latest() 获取最新值（不排队）
data = input_port.recv_latest()
if data is not None:
    process(data)

# ❌ 错误：假设能排队接收
data = input_port.recv()  # 不存在此方法
```

#### 5. 日志记录

```python
# ✅ 正确：使用分级日志
sdk.logger.info("节点启动")      # 正常运行信息
sdk.logger.debug("详细信息")      # 调试信息
sdk.logger.warning("警告信息")    # 潜在问题
sdk.logger.error("错误信息")      # 错误条件

# ❌ 错误：使用 print
print("数据")  # 不符合规范，日志级别混乱
```

## 数据传输协议

### 消息格式

所有数据通过 Unix Domain Socket 以 JSON 格式传输。

```
[4字节长度前缀][JSON消息体]
```

**示例**：

```
长度前缀: 0x00 0x00 0x00 0x78 (120 bytes)
消息体: {"timestamp": 1703024780.123, "seq": 1, "latitude": 39.9042, ...}
```

### 数据类型规范

#### JSON 类型（推荐）

```python
# GPS 数据
{
    "timestamp": 1703024780.123,
    "seq": 1,
    "latitude": 39.9042,
    "longitude": 116.4074,
    "altitude": 50.0,
    "accuracy": 0.05
}

# 控制指令
{
    "timestamp": 1703024780.5,
    "seq": 1,
    "velocity": 1.5,
    "steering_angle": 0.2,
    "mode": "pure_pursuit"
}

# 路径数据
{
    "timestamp": 1703024780.5,
    "seq": 1,
    "path": [[116.4074, 39.9042], [116.4076, 39.9044], ...],
    "status": "success",
    "length": 150.5
}
```

#### 字段约定

- **timestamp**: 浮点数，秒级时间戳（必需）
- **seq**: 整数，消息序号，从0递增（必需）
- **其他字段**: 根据数据类型定义

### 序列号连续性

节点发送的消息应该有连续的 `seq` 号，用于检测丢包：

```python
seq = 0
while True:
    data = {
        'timestamp': time.time(),
        'seq': seq,
        # ... 其他字段
    }
    output_port.send(data)
    seq += 1  # 每次递增
```

## 参数系统

### 参数类型

| 类型 | YAML 示例 | Python 类型 | 说明 |
|------|---------|-----------|------|
| `string` | `"hello"` | `str` | 字符串 |
| `number` | `3.14` | `float` | 浮点数 |
| `integer` | `10` | `int` | 整数（number 的子集） |
| `bool` | `true` | `bool` | 布尔值 |
| `json` | `{...}` | `dict` | JSON 对象（高级） |

### 参数传递机制

参数通过环境变量传给节点进程：

```
NODE_ID=gps_0
NODE_PARAM_mode=stationary
NODE_PARAM_update_rate_hz=10
NODE_PARAM_start_latitude=39.9042
```

SDK 自动解析这些环境变量：

```python
# SDK 内部处理
mode = sdk.get_param('mode', 'stationary')
# 相当于：
mode = os.environ.get('NODE_PARAM_mode', 'stationary')
```

## 端口系统

### 端口命名规范

- **输入端口**: `snake_case`，使用下划线分隔
- **输出端口**: `snake_case`，使用下划线分隔
- **避免**: 使用大写字母、特殊符号

```yaml
# ✅ 正确
ports:
  inputs:
    - name: gps_fix
    - name: control_input
  outputs:
    - name: processed_data
    - name: status_report

# ❌ 错误
ports:
  inputs:
    - name: GPSFix        # 应该是 gps_fix
    - name: control-input # 应该是 control_input
```

### 端口连接规则

在 YAML 配置中定义端口连接：

```yaml
edges:
  - from: source_node.output_port
    to: target_node.input_port
```

**约束**:
- 一个输出端口可连接到多个输入端口（一对多）
- 一个输入端口只能连接到一个输出端口（多对一）
- 不能形成循环依赖

### 端口类型（未来扩展）

目前所有端口都是 `json` 或 `any` 类型：

```yaml
ports:
  outputs:
    - name: data
      type: json        # 推荐使用 JSON 类型
```

类型检查由框架验证，确保连接兼容。

## 常见错误与修复

### 错误 1: node.yaml 格式错误 ❌

**症状**：运行时报错 `'list' object has no attribute 'items'`

**原因**：params 定义为列表而不是字典

**错误代码**：
```yaml
params:
  - name: param1
    type: string
  - name: param2
    type: number
```

**修复**：
```yaml
params:
  param1:
    type: string
  param2:
    type: number
```

### 错误 2: 端口定义不在 ports 内 ❌

**症状**：节点未正确加载，输入/输出端口无法找到

**错误代码**：
```yaml
inputs:
  - name: input1
outputs:
  - name: output1
```

**修复**：
```yaml
ports:
  inputs:
    - name: input1
  outputs:
    - name: output1
```

### 错误 3: 缺少 entrypoints 的 kind 字段 ❌

**症状**：节点启动失败，`kind` 字段未找到

**错误代码**：
```yaml
entrypoints:
  linux:
    cmd: ["python3", "run.py"]
```

**修复**：
```yaml
entrypoints:
  linux:
    kind: python              # 添加此字段
    cmd: ["python3", "run.py"]
```

### 错误 4: 输入端口 socket 连接失败 ❌

**症状**：`Failed to connect InputPort 'xxx' to /tmp/nodeflow_sockets/nodeflow_node_0.xxx.in`

**原因**：EnvBuilder 创建了错误的 socket 路径（过时的 bug，已修复）

**说明**：这个 bug 在 2025-12-20 已修复。输入端口现在正确指向上游节点的输出 socket。

### 错误 5: 端口名称不匹配 ❌

**症状**：运行时找不到指定的端口

**错误示例**：
```python
# node.yaml 中
ports:
  outputs:
    - name: gps_fix

# run.py 中
output_port = sdk.create_output_port('gps_data')  # ❌ 名称不匹配
```

**修复**：
```python
output_port = sdk.create_output_port('gps_fix')   # ✅ 与 node.yaml 一致
```

### 错误 6: 循环依赖 ❌

**症状**：拓扑排序失败，无法确定启动顺序

**错误图**：
```
A → B → C → A （形成环）
```

**修复**：检查 edges 配置，确保图是有向无环图 (DAG)

```yaml
edges:
  - from: A.out
    to: B.in
  - from: B.out
    to: C.in
  # ❌ 不要添加 C → A，会形成环
```

### 错误 7: 参数类型转换失败 ❌

**症状**：参数传递不正确，值的类型错误

**错误示例**：
```python
# YAML 中定义为 number
update_rate_hz:
  type: number
  default: 10

# 但在 run.py 中期望字符串
rate = sdk.get_param('update_rate_hz')  # 得到 '10'（字符串）
delay = 1.0 / rate                       # ❌ TypeError
```

**修复**：SDK 自动类型转换，确保类型一致

```python
rate = float(sdk.get_param('update_rate_hz', 10))  # ✅ 显式转换
delay = 1.0 / rate
```

### 示例 1: 简单的源节点（Joystick）

**场景**：模拟游戏手柄输入，生成控制指令

**node.yaml**:
```yaml
name: mock_joystick
version: "1.0"
description: "游戏手柄模拟器"

entrypoints:
  linux:
    kind: python
    cmd: ["python3", "run.py"]

ports:
  inputs: []
  outputs:
    - name: control_input
      type: json
      description: "控制指令"

params:
  mode:
    type: string
    default: "sine"
    description: "模式：constant, sine, random"
  update_rate_hz:
    type: number
    default: 50
    description: "更新频率"
```

**run.py**:
```python
#!/usr/bin/env python3
import time
import math
from pathlib import Path
import sys

sys.path.insert(0, str(Path(__file__).parent.parent.parent))
from sdk.nodeflow_sdk import NodeFlowSDK

def main():
    with NodeFlowSDK(log_level="INFO") as sdk:
        sdk.logger.info(f"Joystick Mock 启动: {sdk.node_id}")

        mode = sdk.get_param('mode', 'sine')
        rate = sdk.get_param('update_rate_hz', 50)

        output_port = sdk.create_output_port('control_input')

        seq = 0
        start_time = time.time()

        try:
            while True:
                elapsed = time.time() - start_time

                # 根据模式生成控制值
                if mode == 'sine':
                    linear = 1.5 * math.sin(2 * math.pi * elapsed / 10)
                    angular = 0.5 * math.cos(2 * math.pi * elapsed / 10)
                else:
                    linear = 1.5
                    angular = 0.0

                data = {
                    'timestamp': time.time(),
                    'seq': seq,
                    'linear_velocity': round(linear, 3),
                    'angular_velocity': round(angular, 3)
                }

                output_port.send(data)

                if seq % rate == 0:
                    sdk.logger.debug(f"发送 #{seq}: 线速度={data['linear_velocity']}")

                seq += 1
                time.sleep(1.0 / rate)

        except KeyboardInterrupt:
            sdk.logger.info("关闭")
        finally:
            sdk.logger.info(f"完成，共 {seq} 个消息")

if __name__ == '__main__':
    main()
```

```python
def main():
    with NodeFlowSDK(log_level="INFO") as sdk:
        sdk.logger.info("路径规划器启动")

        path_length = sdk.get_param('path_length', 10)

        target_input = sdk.create_input_port('target')
        path_output = sdk.create_output_port('path')

        task_count = 0

        try:
            while True:
                # 非阻塞接收
                target = target_input.recv_latest()

                if target is not None:
                    sdk.logger.info(f"收到目标: {target}")

                    # 规划路径（示例）
                    path = [
                        [target['lon'] + 0.001 * i, target['lat'] + 0.001 * i]
                        for i in range(path_length)
                    ]

                    result = {
                        'timestamp': time.time(),
                        'seq': task_count,
                        'path': path,
                        'status': 'success'
                    }

                    path_output.send(result)
                    task_count += 1

                time.sleep(0.1)

        except KeyboardInterrupt:
            sdk.logger.info("关闭")
        finally:
            sdk.logger.info(f"完成 {task_count} 个规划")
```

**run.py** (关键部分):
```python
def main():
    with NodeFlowSDK(log_level="INFO") as sdk:
        sdk.logger.info("数据验证器启动")

        min_freq = sdk.get_param('min_frequency_hz', 5)
        data_input = sdk.create_input_port('data_in')

        total_count = 0
        error_count = 0
        start_time = time.time()

        try:
            while True:
                data = data_input.recv_latest()

                if data:
                    total_count += 1

                    # 验证数据
                    if 'seq' not in data or 'timestamp' not in data:
                        error_count += 1
                        sdk.logger.warning("数据不完整")

                    if total_count % 100 == 0:
                        elapsed = time.time() - start_time
                        freq = total_count / elapsed
                        sdk.logger.info(f"验证: {total_count} 条, 频率={freq:.1f}Hz")

                time.sleep(0.05)

        except KeyboardInterrupt:
            sdk.logger.info("关闭")
        finally:
            elapsed = time.time() - start_time
            avg_freq = total_count / elapsed if elapsed > 0 else 0
            error_rate = (error_count / total_count * 100) if total_count > 0 else 0
            sdk.logger.info(
                f"统计: {total_count} 条消息, "
                f"平均频率={avg_freq:.1f}Hz, "
                f"错误率={error_rate:.1f}%"
            )
```
